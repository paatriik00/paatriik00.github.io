2. index.html 
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Raspberry Pi Temperature Monitor</title>
    <script src="https://cdn.jsdelivr.net/npm/chart.js"></script>
</head>
<body style="background:#0f172a; color:#fff; font-family:sans-serif; padding:20px;">
    <h1>Raspberry Pi Monitor</h1>
    <canvas id="myChart"></canvas>
    <script>
        fetch('temperaturas_rpi.json')
            .then(res => res.json())
            .then(json => {
                const labels = json.data.map(d => `${d.date.slice(5)} ${d.time}`);
                const temps = json.data.map(d => d.temperature_celsius);
                new Chart(document.getElementById('myChart'), {
                    type: 'line',
                    data: {
                        labels: labels,
                        datasets: [{ label: 'Temp (°C)', data: temps, borderColor: '#ff4757', fill: true }]
                    }
                });
            });
    </script>
</body>
</html>
